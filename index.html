async function unmuteCameraFeed() { if (isActivated) return; statusText.innerText = "//STATUS: HARDWARE BOOTING"; actionText.innerText = "//OPENING OPTICAL CHANNELS..."; try { const stream = await navigator.mediaDevices.getUserMedia({ video: { width: 640, height: 480 }, audio: false }); video.srcObject = stream; isActivated = true; bootBtn.style.display = "none"; } catch (err) { console.error("Camera error:", err); statusText.innerText = "//STATUS: CAMERA ERROR"; actionText.innerText = "//CAMERA ACCESS DENIED OR UNAVAILABLE"; bootBtn.style.display = "block"; } }
